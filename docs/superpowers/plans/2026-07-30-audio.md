# M4 Audio & Music Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a full sound layer — looping music/ambience + one-shot effects for every key moment — driven by one client `Sound` module with all audio IDs in one editable table, plus a saved 🔊/🔇 mute toggle.

**Architecture:** One client module `src/client/Sound.luau` (sibling to `Fx.luau`) owns a `CUES` config table, builds reusable `Sound` templates under `SoundService`, and exposes `Sound.play("Cue")`. Other client files call only `Sound.play(...)`. The mute toggle mirrors the existing `DarkMode` pattern exactly (a HUD button + `PlayerData.soundMuted` + a `SetSoundMuted` remote + the `RequestInitialState`/`PlayerStateUpdated` pull). All sounds are 2D/local. Unset ids (`rbxassetid://0`) are silent no-ops.

**Tech Stack:** Roblox / Luau, Rojo, `SoundService`.

**Spec:** `docs/superpowers/specs/2026-07-30-audio-design.md`

## Global Constraints

- **Client-only playback.** Server only persists the mute boolean (one field + one remote), exactly like `SetDarkMode`.
- **Unset = silent, never errors.** Any cue whose id normalizes to empty or `rbxassetid://0` makes `Sound.play` a no-op. The game must run with every slot empty.
- **Rapid repeats overlap cleanly.** One-shots clone-and-play so mashing (clicks/taps) never cuts off or leaks instances.
- **Mute persists + covers everything** (music, ambience, effects). Default = sound ON.
- **Mobile:** the mute button is a HUD button (46×46) beside the DarkMode toggle; 2D sounds play fine on phones.
- **Layer, don't replace.** Hooks are single `Sound.play(...)` calls added at existing sites — no existing logic changes.
- **Testing = `rojo build` compile check + Studio playtest.** Audio/UI can't be unit-tested; verify by playtest. With ids unset, a playtest verifies "no errors + button/behaviour", not actual sound.
- Compile check after each task: `./rojo-bin/rojo build default.project.json -o /tmp/audio.rbxl` (expect "Built project to audio.rbxl").

- [ ] **Step 4: Create `src/client/Sound.luau`** with exactly:

```lua
local SoundService = game:GetService("SoundService")
local Remotes = require(game.ReplicatedStorage.Shared.Remotes)

local Sound = {}

-- Every game sound in ONE place. Fill ids from Studio -> Toolbox -> Audio.
-- Paste the full "rbxassetid://XXXX" or just the number; an id of 0 (or blank)
-- is "unset" and plays nothing (no errors), so the game runs while slots fill.
local CUES = {
	Music       = { id = "rbxassetid://0", volume = 0.30, looped = true },
	Ambient     = { id = "rbxassetid://0", volume = 0.15, looped = true },
	Click       = { id = "rbxassetid://0", volume = 0.50 },
	PanelOpen   = { id = "rbxassetid://0", volume = 0.45 },
	Error       = { id = "rbxassetid://0", volume = 0.45 },
	Teleport    = { id = "rbxassetid://0", volume = 0.50 },
	Cash        = { id = "rbxassetid://0", volume = 0.50 },
	Purchase    = { id = "rbxassetid://0", volume = 0.60 },
	Hire        = { id = "rbxassetid://0", volume = 0.55 },
	DailyReward = { id = "rbxassetid://0", volume = 0.60 },
	RoundStart  = { id = "rbxassetid://0", volume = 0.40 },
	MinigameTap = { id = "rbxassetid://0", volume = 0.40 },
	Correct     = { id = "rbxassetid://0", volume = 0.50 },
	Wrong       = { id = "rbxassetid://0", volume = 0.40 },
	Shipped     = { id = "rbxassetid://0", volume = 0.60 },
	TrendHit    = { id = "rbxassetid://0", volume = 0.70 },
	Milestone   = { id = "rbxassetid://0", volume = 0.70 },
	CaseSpin    = { id = "rbxassetid://0", volume = 0.45 },
	CaseReveal  = { id = "rbxassetid://0", volume = 0.65 },
	Rebirth     = { id = "rbxassetid://0", volume = 0.70 },
}

local function normalizeId(id)
	if id == nil or id == "" then
		return ""
	end
	if string.match(id, "^%d+$") then
		return "rbxassetid://" .. id
	end
	return id
end

local function isUnset(id)
	local n = normalizeId(id)
	return n == "" or n == "rbxassetid://0"
end

local folder
local templates = {}
local muted = false
local playerState

function Sound.isMuted()
	return muted
end

function Sound.play(cueName)
	if muted then
		return
	end
	local spec = CUES[cueName]
	local t = templates[cueName]
	if not spec or not t or spec.looped or isUnset(spec.id) then
		return
	end
	-- clone so rapid repeats overlap instead of cutting each other off
	local s = t:Clone()
	s.Parent = folder
	s:Play()
	s.Ended:Connect(function()
		s:Destroy()
	end)
	task.delay(6, function()
		if s and s.Parent then
			s:Destroy()
		end
	end)
end

local function startBeds()
	for name, spec in pairs(CUES) do
		if spec.looped and not isUnset(spec.id) then
			local t = templates[name]
			if t and not t.IsPlaying then
				t:Play()
			end
		end
	end
end

local function stopBeds()
	for name, spec in pairs(CUES) do
		if spec.looped then
			local t = templates[name]
			if t and t.IsPlaying then
				t:Stop()
			end
		end
	end
end

function Sound.setMuted(on, persist)
	muted = on and true or false
	if muted then
		stopBeds()
	else
		startBeds()
	end
	if persist then
		if playerState then
			playerState.soundMuted = muted
		end
		Remotes.SetSoundMuted:FireServer(muted)
	end
end

function Sound.init(player, state)
	playerState = state
	folder = Instance.new("Folder")
	folder.Name = "GameAudio"
	folder.Parent = SoundService
	for name, spec in pairs(CUES) do
		local s = Instance.new("Sound")
		s.Name = name
		s.SoundId = isUnset(spec.id) and "" or normalizeId(spec.id)
		s.Volume = spec.volume
		s.Looped = spec.looped == true
		s.Parent = folder
		templates[name] = s
	end

	-- 🔊 / 🔇 toggle button, just below the DarkMode toggle (top-left).
	local pg = player:WaitForChild("PlayerGui")
	local gui = Instance.new("ScreenGui")
	gui.Name = "SoundGui"
	gui.ResetOnSpawn = false
	gui.DisplayOrder = 30
	gui.Parent = pg
	local btn = Instance.new("TextButton")
	btn.Name = "SoundToggle"
	btn.Size = UDim2.fromOffset(46, 46)
	btn.Position = UDim2.fromOffset(16, 70)
	btn.BackgroundColor3 = Color3.fromRGB(120, 90, 235)
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.Font = Enum.Font.FredokaOne
	btn.TextSize = 22
	btn.Text = "\240\159\148\138" -- 🔊
	btn.Parent = gui
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(1, 0)
	c.Parent = btn

	local function refreshIcon()
		btn.Text = muted and "\240\159\148\135" or "\240\159\148\138" -- 🔇 / 🔊
	end
	btn.MouseButton1Click:Connect(function()
		Sound.setMuted(not muted, true)
		refreshIcon()
	end)

	startBeds()

	-- Load the saved preference (mirror DarkMode): first state we receive.
	local loaded = false
	Remotes.PlayerStateUpdated.OnClientEvent:Connect(function(data)
		if not loaded then
			loaded = true
			if data.soundMuted then
				Sound.setMuted(true, false)
				refreshIcon()
			end
		end
	end)
	Remotes.RequestInitialState:FireServer()
end

return Sound
```
