# Personal Home Room (Round 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. The final grid-placement client UI needs a Studio playtest (interactive) to confirm feel.

**Goal:** Each player gets a private Home: touch a white pad at their studio → teleport into their own big room (bigger inside than the studio looks) → place furniture from a free starter set on a grid (move/rotate/delete), and the layout saves.

**Architecture:** A new server module `HomeService` owns one hidden per-player room high in the sky (offset by plot index), a white Home pad at the studio entrance that teleports the owner in, a Leave pad back, and the placement remotes + persistence. Placement is server-authoritative: the client `HomePanel` shows the inventory + a snapping grid ghost and only *requests* placement; `HomeService` validates (owner, in-bounds, non-overlapping, valid item) and is the sole writer of `PlayerData.homeItems`, rebuilding the room's furniture on entry. Pure grid/catalog logic lives in `GameData` so it is unit-testable.

**Tech Stack:** Roblox / Luau, Rojo, existing `PlayerData` / `StudioModels` / `Remotes` / `PlotManager` / `MobileScale`, `RunTests` (`TestHarness:assertEqual`), character `PivotTo` teleport.

## Global Constraints

- **Truly private:** only the owner is teleported into a room; every place/move/remove remote re-checks `PlotManager.getPlayerPlotIndex(player)` owns the targeted room. No cross-player access.
- **Server-authoritative:** client never writes furniture; it sends requests, server validates (item in `GameData.HomeCatalog`, footprint fully in the grid, cells free, caller owns room) and is the only writer of `PlayerData.homeItems`. Rejected requests silently do nothing.
- **Mobile + desktop:** placement works by tap and click; UI follows `StorePanel` style and is picked up by `MobileScale` (a top-level ScreenGui). Tap targets ≥ 44px.
- **Additive / non-disruptive:** the studio, plots, stations, and existing `decorations`/`DecorPanel` system are untouched. The Home pad is added at the studio entrance.
- **Free in Round 1:** all catalog items placeable, unlimited, no cost/unlock. `GameData.StartingCash` stays `0`.
- **Backward compatible:** `PlayerData.homeItems` defaults `{}` + migration backfill.
- **Grid facts (verbatim):** `ROOM_W = 64`, `ROOM_D = 48`, `CELL = 4` → grid `16 × 12` cells; wall height `20`; rooms in the sky at `HOME_SKY_Y = 10000`, spaced `HOME_SPACING = 400` on Z by plot index; rotations are 90° steps only; grid coords are integer `(gx, gz)` 0-indexed from the room's `-X,-Z` corner.
- **Reuse detailed assets; no new Blender this round** — starter catalog uses existing `assets/studio/*.rbxm` where one fits, plus simple procedural pieces flagged for later upgrade.
- **Verification:** `./rojo-bin/rojo build default.project.json -o /tmp/x.rbxl` after each change + Studio playtest. Unit tests only for the pure `GameData` helpers.

```lua
local _homeById = {}
for _, it in ipairs(GameData.HomeCatalog) do _homeById[it.id] = it end
function GameData.getHomeItem(id) return _homeById[id] end

-- Footprint in cells, rotated: a 90/270 yaw swaps width<->depth.
function GameData.homeFootprint(id, yaw)
	local it = _homeById[id]
	if not it then return 1, 1 end
	local w, d = it.cells[1], it.cells[2]
	if yaw == 90 or yaw == 270 then
		return d, w
	end
	return w, d
end

function GameData.homeInBounds(gx, gz, w, d)
	return gx >= 0 and gz >= 0 and (gx + w) <= GameData.Home.cols and (gz + d) <= GameData.Home.rows
end

-- True if the footprint (gx,gz,w,d) overlaps any placed item except items[ignoreIndex].
function GameData.homeOverlaps(items, gx, gz, w, d, ignoreIndex)
	for i, other in ipairs(items) do
		if i ~= ignoreIndex then
			local ow, od = GameData.homeFootprint(other.id, other.yaw)
			if gx < other.gx + ow and gx + w > other.gx and gz < other.gz + od and gz + d > other.gz then
				return true
			end
		end
	end
	return false
end

-- World centre of a footprint anchored at cell (gx,gz), given the room's -X,-Z
-- corner and floor Y. Footprint (w,d) is in cells.
function GameData.homeCellToWorld(cornerX, cornerZ, floorY, gx, gz, w, d)
	local c = GameData.Home.cell
	return Vector3.new(cornerX + gx * c + w * c / 2, floorY, cornerZ + gz * c + d * c / 2)
end
```

```lua
-- Private per-player Home rooms (Round 1). One hidden room per plot index, high
-- in the sky. A white pad at each studio entrance teleports the owner in; a
-- Leave pad returns them. Furniture placement is server-authoritative and saved
-- to PlayerData.homeItems.
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")

local GameData = require(game.ReplicatedStorage.Shared.GameData)
local Remotes = require(game.ReplicatedStorage.Shared.Remotes)
local PlayerData = require(script.Parent.PlayerData)
local PlotManager = require(script.Parent.PlotManager)
local StudioModels = require(script.Parent.StudioModels)

local HomeService = {}

local H = GameData.Home
local NEUTRAL_WALL = Color3.fromRGB(214, 210, 202)
local NEUTRAL_FLOOR = Color3.fromRGB(176, 170, 160)
local ACCENT = Color3.fromRGB(140, 92, 255)

local homesFolder
local roomCorner = {}  -- [index] = Vector3 of the room's -X,-Z floor corner
local roomSpawn = {}   -- [index] = CFrame to drop the player at

local function part(name, size, pos, color, mat, parent)
	local p = Instance.new("Part")
	p.Name, p.Size, p.Position = name, size, pos
	p.Anchored, p.CanCollide = true, true
	p.Color, p.Material = color, mat or Enum.Material.SmoothPlastic
	p.Parent = parent
	return p
end

-- Build the empty room shell for a plot index. Returns the room folder.
local function buildRoom(index)
	local center = Vector3.new(0, H.skyY, index * H.spacing)
	local floorTop = center.Y
	local folder = Instance.new("Folder")
	folder.Name = "Home" .. index
	folder.Parent = homesFolder

	local w, d, wh = H.roomW, H.roomD, H.wallH
	part("Floor", Vector3.new(w, 1, d), center + Vector3.new(0, -0.5, 0), NEUTRAL_FLOOR, Enum.Material.WoodPlanks, folder)
	part("Ceiling", Vector3.new(w, 1, d), center + Vector3.new(0, wh + 0.5, 0), NEUTRAL_WALL, Enum.Material.SmoothPlastic, folder)
	part("WallN", Vector3.new(w, wh, 1), center + Vector3.new(0, wh / 2, -d / 2), NEUTRAL_WALL, nil, folder)
	part("WallS", Vector3.new(w, wh, 1), center + Vector3.new(0, wh / 2, d / 2), NEUTRAL_WALL, nil, folder)
	part("WallW", Vector3.new(1, wh, d), center + Vector3.new(-w / 2, wh / 2, 0), NEUTRAL_WALL, nil, folder)
	local wallE = part("WallE", Vector3.new(1, wh, d), center + Vector3.new(w / 2, wh / 2, 0), NEUTRAL_WALL, nil, folder)
	-- a window strip on the east wall (glass, cosmetic)
	local win = part("Window", Vector3.new(1.2, wh * 0.5, d * 0.6), center + Vector3.new(w / 2, wh * 0.55, 0), Color3.fromRGB(150, 195, 215), Enum.Material.Glass, folder)
	win.Transparency = 0.5

	-- Leave pad near the -Z,-X corner.
	local leave = part("LeavePad", Vector3.new(6, 0.4, 6), center + Vector3.new(-w / 2 + 5, 0.2, -d / 2 + 5), ACCENT, Enum.Material.Neon, folder)
	leave.CanCollide = false
	local lg = Instance.new("SurfaceGui"); lg.Face = Enum.NormalId.Top; lg.Parent = leave
	local ll = Instance.new("TextLabel"); ll.Size = UDim2.fromScale(1,1); ll.BackgroundTransparency = 1
	ll.Text = "\240\159\154\170 Leave"; ll.TextColor3 = Color3.new(1,1,1); ll.Font = Enum.Font.FredokaOne; ll.TextScaled = true; ll.Parent = lg

	-- furniture goes in its own subfolder we can clear + rebuild.
	local furn = Instance.new("Folder"); furn.Name = "Furniture"; furn.Parent = folder

	roomCorner[index] = center + Vector3.new(-w / 2, 0, -d / 2)  -- floor corner, y = floorTop
	roomCorner[index] = Vector3.new(roomCorner[index].X, floorTop, roomCorner[index].Z)
	roomSpawn[index] = CFrame.new(center + Vector3.new(0, 4, d / 2 - 6), center + Vector3.new(0, 4, 0))

	-- Leave pad -> teleport back to studio entrance.
	leave.Touched:Connect(function(hit)
		local plr = Players:GetPlayerFromCharacter(hit.Parent)
		if plr and PlotManager.getPlayerPlotIndex(plr) == index then
			local cf = PlotManager.getStudioEntranceCFrame(index)
			if cf and plr.Character then plr.Character:PivotTo(cf) end
		end
	end)
	return folder
end

-- Build one item's model at its grid cell (Blender asset or procedural).
local function buildItem(furn, index, entry)
	local it = GameData.getHomeItem(entry.id)
	if not it then return end
	local w, d = GameData.homeFootprint(entry.id, entry.yaw)
	local corner = roomCorner[index]
	local worldW = w * H.cell
	local pos = GameData.homeCellToWorld(corner.X, corner.Z, corner.Y, entry.gx, entry.gz, w, d)
	local cf = CFrame.new(pos) * CFrame.Angles(0, math.rad(entry.yaw or 0), 0)
	local model
	if it.asset and StudioModels.has(it.asset) then
		model = StudioModels.place(it.asset, furn, cf, { name = "HomeItem", targetWidth = worldW - 0.5 })
		if model then
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then p.Anchored = true end
			end
		end
	else
		-- procedural fallback: a simple labelled block sized to the footprint.
		local m = Instance.new("Model"); m.Name = "HomeItem"; m.Parent = furn
		local body = part("Body", Vector3.new(worldW - 0.6, 2.5, d * H.cell - 0.6), pos + Vector3.new(0, 1.25, 0), Color3.fromRGB(150, 150, 165), nil, m)
		body.CFrame = cf * CFrame.new(0, 1.25, 0)
		model = m
	end
	if model then
		model:SetAttribute("gx", entry.gx); model:SetAttribute("gz", entry.gz)
	end
	return model
end

function HomeService.rebuildHome(player)
	local index = PlotManager.getPlayerPlotIndex(player)
	if not index then return end
	local folder = homesFolder:FindFirstChild("Home" .. index)
	if not folder then return end
	local furn = folder:FindFirstChild("Furniture")
	if not furn then return end
	furn:ClearAllChildren()
	local data = PlayerData.get(player)
	if not data then return end
	for _, entry in ipairs(data.homeItems) do
		pcall(buildItem, furn, index, entry)
	end
end

local function enter(player)
	local index = PlotManager.getPlayerPlotIndex(player)
	if not index or not roomSpawn[index] then return end
	HomeService.rebuildHome(player)
	if player.Character then player.Character:PivotTo(roomSpawn[index]) end
end

function HomeService.build()
	homesFolder = Workspace:FindFirstChild("Homes")
	if homesFolder then homesFolder:Destroy() end
	homesFolder = Instance.new("Folder"); homesFolder.Name = "Homes"; homesFolder.Parent = Workspace
	for index = 1, 4 do
		buildRoom(index)
		-- Home pad at the studio entrance for this plot.
		local origin = PlotManager.getPlotOrigin(index)
		if origin then
			local padPos = origin + Vector3.new(0, 1.2, 45 - 16 - 9) -- near studio entrance apron
			local pad = part("HomePad" .. index, Vector3.new(6, 0.4, 6), padPos, Color3.fromRGB(245, 245, 250), Enum.Material.Neon, Workspace)
			pad.CanCollide = false
			local sg = Instance.new("SurfaceGui"); sg.Face = Enum.NormalId.Top; sg.Parent = pad
			local lbl = Instance.new("TextLabel"); lbl.Size = UDim2.fromScale(1,1); lbl.BackgroundTransparency = 1
			lbl.Text = "\240\159\143\160 My Home"; lbl.TextColor3 = ACCENT; lbl.Font = Enum.Font.FredokaOne; lbl.TextScaled = true; lbl.Parent = sg
			pad:SetAttribute("plotIndex", index)
			pad.Touched:Connect(function(hit)
				local plr = Players:GetPlayerFromCharacter(hit.Parent)
				if plr and PlotManager.getPlayerPlotIndex(plr) == index then
					enter(plr)
				end
			end)
		end
	end
end

function HomeService.start()
	Remotes.RequestEnterHome.OnServerEvent:Connect(function(player) enter(player) end)
	Remotes.RequestLeaveHome.OnServerEvent:Connect(function(player)
		local index = PlotManager.getPlayerPlotIndex(player)
		local cf = index and PlotManager.getStudioEntranceCFrame(index)
		if cf and player.Character then player.Character:PivotTo(cf) end
	end)

	Remotes.RequestPlaceHomeItem.OnServerEvent:Connect(function(player, id, gx, gz, yaw)
		local index = PlotManager.getPlayerPlotIndex(player)
		local data = PlayerData.get(player)
		if not index or not data then return end
		if not GameData.getHomeItem(id) then return end
		gx, gz, yaw = math.floor(tonumber(gx) or -1), math.floor(tonumber(gz) or -1), tonumber(yaw) or 0
		if yaw ~= 0 and yaw ~= 90 and yaw ~= 180 and yaw ~= 270 then return end
		local w, d = GameData.homeFootprint(id, yaw)
		if not GameData.homeInBounds(gx, gz, w, d) then return end
		if GameData.homeOverlaps(data.homeItems, gx, gz, w, d, nil) then return end
		table.insert(data.homeItems, { id = id, gx = gx, gz = gz, yaw = yaw })
		HomeService.rebuildHome(player)
		Remotes.HomeStateUpdated:FireClient(player, data.homeItems)
	end)

	Remotes.RequestMoveHomeItem.OnServerEvent:Connect(function(player, i, gx, gz, yaw)
		local index = PlotManager.getPlayerPlotIndex(player)
		local data = PlayerData.get(player)
		if not index or not data then return end
		i = math.floor(tonumber(i) or -1)
		local entry = data.homeItems[i]
		if not entry then return end
		gx, gz, yaw = math.floor(tonumber(gx) or -1), math.floor(tonumber(gz) or -1), tonumber(yaw) or 0
		if yaw ~= 0 and yaw ~= 90 and yaw ~= 180 and yaw ~= 270 then return end
		local w, d = GameData.homeFootprint(entry.id, yaw)
		if not GameData.homeInBounds(gx, gz, w, d) then return end
		if GameData.homeOverlaps(data.homeItems, gx, gz, w, d, i) then return end
		entry.gx, entry.gz, entry.yaw = gx, gz, yaw
		HomeService.rebuildHome(player)
		Remotes.HomeStateUpdated:FireClient(player, data.homeItems)
	end)

	Remotes.RequestRemoveHomeItem.OnServerEvent:Connect(function(player, i)
		local data = PlayerData.get(player)
		if not data then return end
		i = math.floor(tonumber(i) or -1)
		if not data.homeItems[i] then return end
		table.remove(data.homeItems, i)
		HomeService.rebuildHome(player)
		Remotes.HomeStateUpdated:FireClient(player, data.homeItems)
	end)
end

return HomeService
```

### Task 7: Client HomePanel — inventory + grid ghost + place/move/delete

**Files:**
- Create: `src/client/HomePanel.luau`
- Modify: `src/client/UI.luau` (require + `HomePanel.init(player)` alongside the other panel inits)

**Interfaces:** Consumes `GameData.HomeCatalog/Home/homeFootprint/homeCellToWorld` and the home remotes. Produces `HomePanel.init(player)`.

- [ ] **Step 1: Create `src/client/HomePanel.luau`** — an inventory panel + a raycast-to-floor grid ghost that snaps to cells and requests placement. Only usable while inside the Home (detected by the player's Y being near `GameData.Home.skyY`).

```lua
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local Remotes = require(game.ReplicatedStorage.Shared.Remotes)
local GameData = require(game.ReplicatedStorage.Shared.GameData)

local HomePanel = {}
local H = GameData.Home

local plr, gui, listFrame, ghost, selectedId, curYaw, myItems
local placing = false

local function inHome()
	local ch = plr.Character
	local root = ch and ch:FindFirstChild("HumanoidRootPart")
	return root ~= nil and math.abs(root.Position.Y - H.skyY) < 200
end

-- The room corner for THIS player = derived from their plot index is server-side;
-- the client finds it from the room floor it is standing on.
local function roomCorner()
	local ch = plr.Character
	local root = ch and ch:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	-- room center Z ~ round(root.Z / spacing)*spacing; corner = center - (roomW/2, 0, roomD/2)
	local index = math.floor(root.Position.Z / H.spacing + 0.5)
	local centerZ = index * H.spacing
	return Vector3.new(-H.roomW / 2, H.skyY, centerZ - H.roomD / 2)
end

local function clearGhost()
	if ghost then ghost:Destroy(); ghost = nil end
	placing = false
	selectedId = nil
end

local function makeGhost(id)
	clearGhost()
	selectedId = id; curYaw = 0; placing = true
	local w, d = GameData.homeFootprint(id, curYaw)
	ghost = Instance.new("Part")
	ghost.Anchored = true; ghost.CanCollide = false; ghost.Transparency = 0.5
	ghost.Color = Color3.fromRGB(120, 220, 140)
	ghost.Size = Vector3.new(w * H.cell - 0.4, 1, d * H.cell - 0.4)
	ghost.Parent = workspace
end

-- Snap the ghost to the grid cell under the mouse/touch, colour green/red.
local function updateGhost()
	if not placing or not ghost then return end
	local corner = roomCorner()
	if not corner then return end
	local cam = workspace.CurrentCamera
	local pos = UserInputService:GetMouseLocation()
	local ray = cam:ViewportPointToRay(pos.X, pos.Y)
	local rc = RaycastParams.new()
	rc.FilterType = Enum.RaycastFilterType.Exclude
	rc.FilterDescendantsInstances = { ghost, plr.Character }
	local hit = workspace:Raycast(ray.Origin, ray.Direction * 2000, rc)
	if not hit then return end
	local w, d = GameData.homeFootprint(selectedId, curYaw)
	local gx = math.floor((hit.Position.X - corner.X) / H.cell)
	local gz = math.floor((hit.Position.Z - corner.Z) / H.cell)
	gx = math.clamp(gx, 0, H.cols - w)
	gz = math.clamp(gz, 0, H.rows - d)
	ghost.Size = Vector3.new(w * H.cell - 0.4, 1, d * H.cell - 0.4)
	local wc = GameData.homeCellToWorld(corner.X, corner.Z, H.skyY + 0.6, gx, gz, w, d)
	ghost.CFrame = CFrame.new(wc) * CFrame.Angles(0, math.rad(curYaw), 0)
	local free = not GameData.homeOverlaps(myItems or {}, gx, gz, w, d, nil)
	ghost.Color = free and Color3.fromRGB(120, 220, 140) or Color3.fromRGB(230, 90, 90)
	ghost:SetAttribute("gx", gx); ghost:SetAttribute("gz", gz)
end

local function confirmPlace()
	if not placing or not ghost then return end
	local gx, gz = ghost:GetAttribute("gx"), ghost:GetAttribute("gz")
	if gx and gz then
		Remotes.RequestPlaceHomeItem:FireServer(selectedId, gx, gz, curYaw)
	end
	clearGhost()
end

local function addCard(item)
	local b = Instance.new("TextButton")
	b.Size = UDim2.new(1, 0, 0, 40)
	b.BackgroundColor3 = Color3.fromRGB(238, 238, 246)
	b.Text = "  " .. item.icon .. "  " .. item.name
	b.TextXAlignment = Enum.TextXAlignment.Left
	b.TextColor3 = Color3.fromRGB(30, 33, 48)
	b.Font = Enum.Font.GothamMedium; b.TextSize = 16
	b.LayoutOrder = 1
	local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 8); c.Parent = b
	b.MouseButton1Click:Connect(function() makeGhost(item.id) end)
	b.Parent = listFrame
end

function HomePanel.toggle()
	if not gui then return end
	gui.Enabled = not gui.Enabled
end

function HomePanel.init(player)
	plr = player
	myItems = {}
	local pg = player:WaitForChild("PlayerGui")
	gui = Instance.new("ScreenGui")
	gui.Name = "HomePanelGui"; gui.ResetOnSpawn = false; gui.DisplayOrder = 45; gui.Enabled = false
	gui.Parent = pg

	-- A "Furniture" button, only visible while inside the Home.
	local openBtn = Instance.new("TextButton")
	openBtn.Name = "FurnitureButton"
	openBtn.Size = UDim2.fromOffset(150, 44)
	openBtn.Position = UDim2.new(0, 16, 1, -60)
	openBtn.AnchorPoint = Vector2.new(0, 1)
	openBtn.BackgroundColor3 = Color3.fromRGB(140, 92, 235)
	openBtn.TextColor3 = Color3.new(1, 1, 1)
	openBtn.Font = Enum.Font.FredokaOne; openBtn.TextSize = 18
	openBtn.Text = "\240\159\170\145 Furniture"
	openBtn.Visible = false
	local oc = Instance.new("UICorner"); oc.CornerRadius = UDim.new(0, 10); oc.Parent = openBtn
	openBtn.Parent = pg:FindFirstChild("HomePanelGui")
	openBtn.MouseButton1Click:Connect(function() gui.Enabled = not gui.Enabled end)

	-- Panel with the catalog list + a Leave button.
	local panel = Instance.new("Frame")
	panel.AnchorPoint = Vector2.new(0, 0.5); panel.Position = UDim2.new(0, 16, 0.5, 0)
	panel.Size = UDim2.fromOffset(240, 420); panel.BackgroundColor3 = Color3.fromRGB(255,255,255)
	panel.Parent = gui
	local pc = Instance.new("UICorner"); pc.CornerRadius = UDim.new(0, 14); pc.Parent = panel
	local title = Instance.new("TextLabel")
	title.Size = UDim2.new(1, -20, 0, 40); title.Position = UDim2.fromOffset(12, 8)
	title.BackgroundTransparency = 1; title.Text = "\240\159\170\145 Furniture"
	title.TextColor3 = Color3.fromRGB(30,33,48); title.Font = Enum.Font.FredokaOne
	title.TextSize = 22; title.TextXAlignment = Enum.TextXAlignment.Left; title.Parent = panel

	listFrame = Instance.new("ScrollingFrame")
	listFrame.Position = UDim2.fromOffset(12, 52); listFrame.Size = UDim2.new(1, -24, 1, -110)
	listFrame.BackgroundTransparency = 1; listFrame.ScrollBarThickness = 6
	listFrame.CanvasSize = UDim2.new(); listFrame.AutomaticCanvasSize = Enum.AutomaticSize.Y
	listFrame.Parent = panel
	local ll = Instance.new("UIListLayout"); ll.Padding = UDim.new(0, 6); ll.Parent = listFrame
	for _, item in ipairs(GameData.HomeCatalog) do addCard(item) end

	local leaveBtn = Instance.new("TextButton")
	leaveBtn.Size = UDim2.new(1, -24, 0, 40); leaveBtn.Position = UDim2.new(0, 12, 1, -48)
	leaveBtn.BackgroundColor3 = Color3.fromRGB(230, 90, 90); leaveBtn.TextColor3 = Color3.new(1,1,1)
	leaveBtn.Font = Enum.Font.FredokaOne; leaveBtn.TextSize = 18; leaveBtn.Text = "\240\159\154\170 Leave Home"
	local lc = Instance.new("UICorner"); lc.CornerRadius = UDim.new(0, 10); lc.Parent = leaveBtn
	leaveBtn.Parent = panel
	leaveBtn.MouseButton1Click:Connect(function()
		Remotes.RequestLeaveHome:FireServer(); gui.Enabled = false; clearGhost()
	end)

	-- Track my placed items (for the ghost's red/green overlap check).
	Remotes.HomeStateUpdated.OnClientEvent:Connect(function(items) myItems = items end)

	-- Show/hide the Furniture button based on being inside the Home.
	RunService.RenderStepped:Connect(function()
		local here = inHome()
		openBtn.Visible = here
		if not here and gui.Enabled then gui.Enabled = false; clearGhost() end
		updateGhost()
	end)

	-- Place on click/tap; rotate with R.
	UserInputService.InputBegan:Connect(function(input, gp)
		if gp then return end
		if placing and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
			confirmPlace()
		elseif placing and input.KeyCode == Enum.KeyCode.R then
			curYaw = (curYaw + 90) % 360
		end
	end)
end

return HomePanel
```
